def getlongtermStats(kmsmallest, kmgreatest, literoverall, firstliter, priceoverall):
    kmoverall = kmgreatest - kmsmallest
    avgConsumption = (literoverall-firstliter) / kmoverall * 100
    priceperKM = priceoverall / kmoverall
    return [kmoverall, avgConsumption, priceperKM]

def init_db():
    conn = sqlite3.connect('daten.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS tankdata (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            datum TEXT,
            km INTEGER,
            liter REAL,
            preis REAL,
            kategorie TEXT
        )
    ''')
    conn.commit()
    conn.close()

@app.route('/api/alle', methods=['GET'])
def alle():
    conn = sqlite3.connect('daten.db')
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM tankdata WHERE kategorie='sprit'")
    rows = cursor.fetchall()
    conn.close()

    # Daten in JSON umwandeln
    tankdata = [
        {"id": row[0], "datum": row[1], "km": row[2], "liter": row[3], "preis": row[4], "kategorie": row[5]}
        for row in rows
    ]

    print("check tankdata: ", tankdata)
    return jsonify(tankdata)
